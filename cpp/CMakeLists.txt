cmake_minimum_required(VERSION 3.16)
project(abstraction_model_api VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT TARGET abstraction::download_request)
    find_package(abstraction_download_request CONFIG QUIET)
endif()
if(NOT TARGET abstraction::download_request AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-download/cpp/CMakeLists.txt")
    set(ABSTRACTION_DOWNLOAD_BUILD_LEGACY OFF)
    add_subdirectory(../../abstraction-download/cpp abstraction_download_request-build)
    unset(ABSTRACTION_DOWNLOAD_BUILD_LEGACY)
endif()
if(NOT TARGET abstraction::download_request)
    message(FATAL_ERROR "Install abstraction_download_request or provide sibling abstraction-download. Nothing is fetched.")
endif()
add_library(abstraction_model_api INTERFACE)
add_library(abstraction::model_api ALIAS abstraction_model_api)
set_target_properties(abstraction_model_api PROPERTIES EXPORT_NAME model_api)
target_compile_features(abstraction_model_api INTERFACE cxx_std_17)
target_link_libraries(abstraction_model_api INTERFACE abstraction::download_request)
target_include_directories(abstraction_model_api INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS abstraction_model_api EXPORT abstraction_model_apiTargets)
install(DIRECTORY abstraction/model/api DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/model)
set(config_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_model_api)
install(EXPORT abstraction_model_apiTargets NAMESPACE abstraction:: DESTINATION ${config_dir})
configure_package_config_file(cmake/abstraction_model_apiConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/abstraction_model_apiConfig.cmake INSTALL_DESTINATION ${config_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/abstraction_model_apiConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_model_apiConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/abstraction_model_apiConfigVersion.cmake DESTINATION ${config_dir})

option(ABSTRACTION_MODEL_BUILD_CLIENT "Build shared IPC model client" ON)
if(ABSTRACTION_MODEL_BUILD_CLIENT)
    if(NOT TARGET abstraction::ipc)
        find_package(abstraction_ipc CONFIG QUIET)
    endif()
    if(NOT TARGET abstraction::ipc AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-identity/cpp/CMakeLists.txt")
        add_subdirectory(../../abstraction-identity/cpp abstraction_ipc-build)
    endif()
    if(NOT TARGET abstraction::ipc)
        message(FATAL_ERROR "Install abstraction_ipc or provide sibling abstraction-identity. Nothing is fetched.")
    endif()
    add_library(abstraction_model_client INTERFACE)
    add_library(abstraction::model_client ALIAS abstraction_model_client)
    set_target_properties(abstraction_model_client PROPERTIES EXPORT_NAME model_client)
    target_include_directories(abstraction_model_client INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(abstraction_model_client INTERFACE abstraction::model_api abstraction::ipc)
    install(TARGETS abstraction_model_client EXPORT abstraction_modelTargets)
    install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    set(client_config_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_model)
    install(EXPORT abstraction_modelTargets NAMESPACE abstraction:: DESTINATION ${client_config_dir})
    configure_package_config_file(cmake/abstraction_modelConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/abstraction_modelConfig.cmake INSTALL_DESTINATION ${client_config_dir})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_modelConfig.cmake DESTINATION ${client_config_dir})
endif()
